\subsection{电流}
对于导线上一点的电流, 可以认为其表示为
\begin{equation*}
    I\bm{e_l}=I\frac{\mathrm{d}\bm{l}}{\mathrm{d}l}.
\end{equation*}

定义\textbf{(线) 电流元}为 $I\mathrm{d}\bm{l}$. 同时可以得到\textbf{(体) 电流密度}为
\begin{equation}
    I=\int_S\bm{J}\cdot\mathrm{d}\bm{s}.
\end{equation}

\textbf{面电流元}\quad 假设在一个极薄板状导线内取一截面 (可看作极薄的类似线的矩形), 沿 ``线'' 一方向为 $\bm{e_l}$, 导线板面法线方向为 $\bm{e_n}$, 则规定 $\bm{e_t}=\bm{e_n}\times\bm{e_l}$. 则有
\begin{equation}
    \begin{aligned}
        I & =\int_{0}^{l}\bm{J_S}\cdot\mathrm{d}l\bm{e_t}                 \\
          & =\int_{0}^{l}\bm{J_S}\cdot(\bm{e_n}\times\bm{e_l})\mathrm{d}l \\
          & =\int_{0}^{l}\bm{e_l}\cdot(\bm{J_S}\times\bm{e_n})\mathrm{d}l \\
          & =\int_{0}^{l}(\bm{J_S}\times\bm{e_n})\cdot\mathrm{d}\bm{l}.
    \end{aligned}
\end{equation}

以电荷体密度 $\rho$ 和电荷速度 $\bm{v}$ 表示, 电流密度为
\begin{equation}
    \bm{J}=\rho\bm{v}.
\end{equation}

可以证明,
\begin{equation}
    \nabla\cdot\bm{J}=-\frac{\mathrm{d}\rho}{\mathrm{d}t}.
\end{equation}
这即是微分形式的\textbf{电荷守恒定律}.

对于恒定电流, 有
\begin{equation}
    \frac{\partial\rho}{\partial t}=\nabla\cdot\bm{J}=0.
\end{equation}
也即
\begin{equation}
    \oint_S\bm{J}\cdot\mathrm{d}\bm{S}=0.
\end{equation}
