\subsection{平稳随机过程通过线性系统}
时域表示:
\begin{equation}
    v_o(t)=v_i(t)*h(t)=\int_{-\infty}^{\infty}v_i(\tau)h(t-\tau)\mathrm{d}\tau.
\end{equation}

频域表示:
\begin{equation}
    V_o(\omega)=V_i(\omega)H(\omega).
\end{equation}

\textbf{均值}\quad $E[X_o(t)]=aH(0)$.

\textbf{自相关函数}\quad 仍然满足公式 (\ref{eq:3.2 wiener}).

\textbf{功率谱密度}\quad $P_o(\omega)=P_i(\omega)|H(\omega)|^2$.
