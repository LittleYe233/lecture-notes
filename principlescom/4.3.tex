\subsection{恒参或随参信道特性对信号传输的影响}
\subsubsection{恒参信道}
恒参信道是\textit{线性时不变系统}, 其传输特性可以表示为
\begin{equation}
    H(\omega)=|H(\omega)|e^{j\varphi(\omega)}.
\end{equation}

\textbf{理想恒参信道}\quad $H(\omega)=Ke^{-j\omega t_d}$.
\begin{equation}
    \begin{cases}
        |H(\omega)|=K, \\
        \varphi(\omega)=\omega t_d.
    \end{cases}
\end{equation}
